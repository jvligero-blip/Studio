const SH_STUDENTS = "Alunas"; const SH_ATT = "Presenca"; const SH_MK = "Reposicoes"; function doPost(e) { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); if (data.action === "saveAll") { saveStudents(ss, data.students); saveAtt(ss, data.attendance); saveMakeups(ss, data.makeups || []); return ok("saved"); } if (data.action === "loadAll") { return json({ students: loadStudents(ss), attendance: loadAtt(ss), makeups: loadMakeups(ss) }); } return ok("unknown"); } function saveMakeups(ss, list) { const sh = sheet(ss, SH_MK); sh.clearContents(); const rows = [["id","studentId","originDate","originTime","date","time","note"]]; list.forEach(m => rows.push([ m.id, m.studentId, m.originDate, m.originTime, m.date, m.time, m.note || "" ])); sh.getRange(1,1,rows.length,7).setValues(rows); } function loadMakeups(ss) { const sh = ss.getSheetByName(SH_MK); if (!sh) return []; return sh.getDataRange().getValues().slice(1) .filter(r => r[0]) .map(r => ({ id: String(r[0]), studentId: Number(r[1]), originDate: fmtDay(r[2]), originTime: fmtHour(r[3]), date: fmtDay(r[4]), time: fmtHour(r[5]), note: String(r[6]||"") })); } function fmtDay(v) { if (v instanceof Date) return Utilities.formatDate(v, Session.getScriptTimeZone(), "yyyy-MM-dd"); return String(v || ""); } function fmtHour(v) { if (v instanceof Date) return Utilities.formatDate(v, Session.getScriptTimeZone(), "HH:mm"); return String(v || ""); } function json(o) { return ContentService.createTextOutput(JSON.stringify(o)) .setMimeType(ContentService.MimeType.JSON); } function ok(m) { return json({ok:true, msg:m}); } function sheet(ss, n) { return ss.getSheetByName(n) || ss.insertSheet(n); } function saveStudents(ss, students) { const sh = sheet(ss, SH_STUDENTS); sh.clearContents(); const rows = [["id","name","schedules","modality","obs","colorIdx","faces"]]; students.forEach(s => rows.push([ s.id, s.name, JSON.stringify(s.schedules||[]), s.modality||"", s.obs||"", s.colorIdx||0, JSON.stringify(s.faces||[]) ])); sh.getRange(1,1,rows.length,7).setValues(rows); } function loadStudents(ss) { const sh = ss.getSheetByName(SH_STUDENTS); if (!sh) return []; return sh.getDataRange().getValues().slice(1) .filter(r => r[0] !== "" && r[0] !== null) .map(r => ({ id: Number(r[0]), name: r[1], schedules: safeParse(r[2], []), modality: r[3], obs: r[4], colorIdx: Number(r[5]||0), faces: safeParse(r[6], []) })); } function saveAtt(ss, att) { const sh = sheet(ss, SH_ATT); sh.clearContents(); const rows = [["key","status","arrivalTime","auto"]]; Object.keys(att).forEach(k => { const v = att[k]; rows.push([k, v.status, v.arrivalTime||"", v.auto?"1":""]); }); sh.getRange(1,1,rows.length,4).setValues(rows); } function loadAtt(ss) { const sh = ss.getSheetByName(SH_ATT); if (!sh) return {}; const out = {}; sh.getDataRange().getValues().slice(1) .filter(r => r[0]) .forEach(r => { out[r[0]] = { status: r[1], arrivalTime: String(r[2]||""), auto: r[3] === "1" }; }); return out; } function safeParse(v, fb) { try { return JSON.parse(v || "null") || fb; } catch (err) { return fb; } }
